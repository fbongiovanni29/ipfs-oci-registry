OCI IPFS Registry has been deployed!

1. Get the registry URL:
{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  https://{{ $host.host }}
{{- end }}
{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "oci-ipfs-registry.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
  NOTE: It may take a few minutes for the LoadBalancer IP to be available.
  watch kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "oci-ipfs-registry.fullname" . }}
{{- else if contains "ClusterIP" .Values.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "oci-ipfs-registry.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 5000:5000
  echo "Registry available at http://localhost:5000"
{{- end }}

2. Test the registry:
  curl http://localhost:5000/v2/

3. Pull an image through the proxy:
  docker pull localhost:5000/docker.io/library/alpine:latest

4. Push an image:
  docker tag myimage:latest localhost:5000/myimage:latest
  docker push localhost:5000/myimage:latest

NOTE: For Docker to push/pull from HTTP registries, add to insecure-registries:
  {
    "insecure-registries": ["localhost:5000"]
  }

For more information, visit: https://github.com/fbongiovanni29/ipfs-oci-registry
